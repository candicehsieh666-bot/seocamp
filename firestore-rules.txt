rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ───── 作業與回饋 ─────
    match /artifacts/seo-camp-portal-v1/public/data/{collection}/{docId} {

      allow read: if request.auth != null;

      // 新增時一定要帶上自己的身分，不能冒充別人
      allow create: if request.auth != null
                    && collection != 'config'
                    && request.resource.data.uid is string
                    && request.resource.data.uid == request.auth.uid;

      // 只能刪自己送出的
      allow delete: if request.auth != null
                    && collection != 'config'
                    && resource.data.uid == request.auth.uid;

      // 只開放按讚，內容送出後誰都不能竄改
      allow update: if request.auth != null
                    && collection != 'config'
                    && request.resource.data
                         .diff(resource.data)
                         .affectedKeys()
                         .hasOnly(['likedBy']);
    }

    // ───── 學員名單（config）：大家讀得到，只能從 Firebase 後台改 ─────
    match /artifacts/seo-camp-portal-v1/public/data/config/{docId} {
      allow read: if request.auth != null;
      allow write: if false;
    }

    // ───── 公告：同上 ─────
    match /artifacts/seo-camp-portal-v1/public/{docId} {
      allow read: if request.auth != null;
      allow write: if false;
    }
  }
}
